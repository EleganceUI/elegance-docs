import Lang from '../../../components/Lang.astro';

A control's behavior and its look are separate. A `Button` handles clicks, hovering, focus, and the
keyboard; its *style* decides what it looks like. A style applies to every control of its kind
inside the view you set it on, so one modifier can restyle a whole form.

## Built-in styles

| Control | Styles |
| --- | --- |
| `Button` | `ProminentButtonStyle` (default), `BorderedButtonStyle`, `PlainButtonStyle` |
| `Toggle` | `SwitchToggleStyle` (default) |
| `Checkbox` | `BoxCheckboxStyle` (default) |
| `Picker` | `SegmentedPickerStyle` (default), `MenuPickerStyle` |
| `Slider` | `KnobSliderStyle` (default) |
| `NumberInput` | `BoxNumberInputStyle` (default), `PlainNumberInputStyle` |
| `Progress` | `LinearProgressStyle` (default), `CircularProgressStyle` |
| `TextField`, `TextArea` | `RoundedBorderTextFieldStyle` (default), `PlainTextFieldStyle` |
| `popover` | `DefaultPopoverStyle` (default), `PlainPopoverStyle` |
| `tooltip` | `DefaultTooltipStyle` (default), `PlainTooltipStyle` |
| `Menu` | `DefaultMenuStyle` (default), `PlainMenuStyle` |
| Selectable rows (`.selection`) | `HighlightRowStyle` (default), `PlainRowStyle` |

```cpp
Form().button_style(BorderedButtonStyle{});
Toggle("Wi-Fi", wifi).toggle_style(SwitchToggleStyle{{.fills_width = true}});
Checkbox("Remember me", remember).checkbox_style(BoxCheckboxStyle{{.size = 16}});
TextField("Search", query).text_field_style(PlainTextFieldStyle{});
```

```rust
# let (wifi, remember, query) = (State::new(true), State::new(false), State::new(String::new()));
VStack { Button("OK", || {}) }.button_style(BorderedButtonStyle)
Toggle("Wi-Fi", wifi).toggle_style(SwitchToggleStyle::new().fills_width(true))
Checkbox("Remember me", remember).checkbox_style(BoxCheckboxStyle::new().size(16))
TextField("Search", query).text_field_style(PlainTextFieldStyle)
```

## Writing a button style

A style's `make_body` is given the button's *configuration*, and returns what the button looks
like. It runs **once** for each button, so read the button's state inside closures, which update
as it changes:

```cpp
struct PillButtonStyle : ButtonStyle {
    View make_body(const ButtonConfiguration& button) const override {
        const auto tint = Environment<TintKey>();
        return button.label
            .font(Font::headline)
            .foreground_style(Color::white)
            .padding(EdgeInsets::symmetric(18, 8))
            .background([=] { return button.is_pressed() ? tint().mix(Color::black, 0.2f) : tint(); }, Capsule())
            .opacity([=] { return button.is_enabled() ? 1.f : 0.5f; });
    }
};

HStack(Button("Upgrade", upgrade), Button("Later", later)).button_style(PillButtonStyle{});
```

```rust
struct PillButtonStyle;

impl ButtonStyle for PillButtonStyle {
    fn make_body(&self, button: ButtonConfiguration) -> View {
        let tint = Environment::<TintKey>::new();
        let (pressed, enabled) = (button.clone(), button.clone());
        elg! {
            { button.label() }
                .font(.headline)
                .foreground_style(Color::WHITE)
                .padding(horizontal: 18, vertical: 8)
                .background_in(
                    move || if pressed.is_pressed() { tint.get().mix(Color::BLACK, 0.2) } else { tint.get() },
                    Capsule(),
                )
                .opacity(move || if enabled.is_enabled() { 1.0 } else { 0.5 })
        }
        .into_view()
    }
}

# let (upgrade, later) = (|| {}, || {});
let _ = elg! {
    HStack {
        Button("Upgrade", upgrade)
        Button("Later", later)
    }
    .button_style(PillButtonStyle)
};
```

A `ButtonConfiguration` has the button's `label`, and tells you whether it `is_pressed` (held
down with the pointer still over it), `is_hovered`, `is_enabled`, and `is_focused`. `make_body`
can read the [environment](../environment/), as the tint is read above.

<Lang only="cpp">

The state readers are `Signal<bool>`s: values you can read but not change. Call one inside a
closure to follow it, or pass it straight to anything that takes a `Prop<bool>`, like
`If(button.is_hovered, ...)`.

</Lang>
<Lang only="rust">

Copies of a configuration all refer to the same button, so clone it for each closure that needs
it, as above.

</Lang>

## Toggle styles

A `ToggleConfiguration` has the `label`, whether it `is_on`, `toggle()` to flip it, and the same
`is_pressed`, `is_hovered`, `is_enabled`, and `is_focused`. Override `animation()` to change how
it animates when flipped, or return nothing to flip at once:

```cpp
struct StarToggleStyle : ToggleStyle {
    View make_body(const ToggleConfiguration& toggle) const override {
        return HStack(If(toggle.is_on, Image("star.fill").foreground_style(ThemeColor::warning),
                         Image("star").foreground_style(ThemeColor::tertiary)),
                      toggle.label);
    }
};
```

```rust
struct StarToggleStyle;

impl ToggleStyle for StarToggleStyle {
    fn make_body(&self, toggle: ToggleConfiguration) -> View {
        let on = toggle.clone();
        elg! {
            HStack {
                if on.is_on() {
                    Image("star.fill").foreground_style(.warning)
                } else {
                    Image("star").foreground_style(.tertiary)
                }
                { toggle.label() }
            }
        }
        .into_view()
    }
}
```

The control still handles clicks and the keyboard; the style only draws it.

## Checkbox styles

A `CheckboxConfiguration` has the same parts, with `is_checked` instead of `is_on`, and also
`is_mixed`, for a check box that's partly checked:

```cpp
struct RadioLikeCheckboxStyle : CheckboxStyle {
    View make_body(const CheckboxConfiguration& box) const override {
        return HStack({.spacing = 8},
                      Circle()
                          .stroke_border(ThemeColor::outline, 1.5f)
                          .overlay(Circle().fill(ThemeColor::accent).padding(4).opacity([=] {
                              return box.is_checked() || box.is_mixed() ? 1.f : 0.f;
                          }))
                          .frame(18, 18),
                      box.label);
    }
};
```

```rust
struct RadioLikeCheckboxStyle;

impl CheckboxStyle for RadioLikeCheckboxStyle {
    fn make_body(&self, checkbox: CheckboxConfiguration) -> View {
        let state = checkbox.clone();
        elg! {
            HStack(spacing: 8) {
                Circle()
                    .stroke_border(.outline, 1.5)
                    .overlay(
                        Circle()
                            .fill(.accent)
                            .padding(4)
                            .opacity(move || if state.is_checked() || state.is_mixed() { 1.0 } else { 0.0 })
                    )
                    .frame(18, 18)
                { checkbox.label() }
            }
        }
        .into_view()
    }
}
```

A style can't change how a toggle or check box responds to clicks and keys; that's the same
whatever it looks like.

A control made without a label gets an empty one. Check `label.is_empty()` to leave out the space
you'd put beside it.

## Picker styles

A `PickerConfiguration` has the options' `labels`, which one is `selected` (by its position in
the list<Lang only="cpp" inline>, or `labels.size()` while none is</Lang><Lang only="rust" inline>, or `None`</Lang>), and `select(index)` to choose one.
This one is a row of plain text buttons, with the chosen one in the accent color:

```cpp
struct TextRowPickerStyle : PickerStyle {
    View make_body(const PickerConfiguration& picker) const override {
        std::vector<View> options;
        for (std::size_t i = 0; i < picker.labels.size(); ++i) {
            options.push_back(Button(picker.labels[i], [=] { picker.select(i); })
                                  .button_style(PlainButtonStyle{})
                                  .foreground_style([=](const EnvironmentValues& env) {
                                      return env.color(picker.selected() == i ? ThemeColor::accent
                                                                              : ThemeColor::secondary);
                                  }));
        }
        return HStackLayout(VerticalAlignment::center, 12)(std::move(options));
    }
};
```

```rust
struct TextRowPickerStyle;

impl PickerStyle for TextRowPickerStyle {
    fn make_body(&self, picker: PickerConfiguration) -> View {
        let options = picker
            .labels()
            .into_iter()
            .enumerate()
            .map(|(i, label)| {
                let (choose, chosen) = (picker.clone(), picker.clone());
                Button::new(label, move || choose.select(i))
                    .button_style(PlainButtonStyle)
                    .foreground_style(Prop::from_env(move |env| {
                        env.color(if chosen.selected() == Some(i) { ThemeColor::Accent } else { ThemeColor::Secondary })
                    }))
            })
            .collect();
        HStackLayout::new(VerticalAlignment::Center, 12).view(options)
    }
}
```

## Slider styles

A slider's style draws three parts, which the slider puts in place itself, so the style doesn't
need to know how wide the slider is:

- `make_track` is the groove. It gets the slider's width, less the thumb's, and chooses its own
  height.
- `make_fill` covers the track from its start up to the thumb (nothing, unless you draw one).
- `make_thumb` is the handle, shown at its own size, centered on the value.

Each is given a `SliderConfiguration`: where the value is (`fraction`, from 0 to 1), and whether
it `is_pressed` (being dragged), `is_hovered`, `is_enabled`, and `is_focused`. The slider handles
the pointer, the keyboard, and screen readers; the parts only draw.

```cpp
struct BarSliderStyle : SliderStyle {
    View make_track(const SliderConfiguration&) const override {
        return RoundedRectangle(3).fill(ThemeColor::outline).frame({.height = 6});
    }
    View make_fill(const SliderConfiguration&) const override {
        return RoundedRectangle(3).fill(ThemeColor::accent).frame({.height = 6});
    }
    View make_thumb(const SliderConfiguration& slider) const override {
        return RoundedRectangle(2)
            .fill(ThemeColor::accent)
            .frame(4, 16)
            .scale_effect([=] { return slider.is_pressed() ? 1.25f : 1.f; });
    }
};
```

```rust
struct BarSliderStyle;

impl SliderStyle for BarSliderStyle {
    fn make_track(&self, _: SliderConfiguration) -> View {
        elg! { RoundedRectangle(3).fill(.outline).frame(height: 6) }.into_view()
    }
    fn make_fill(&self, _: SliderConfiguration) -> View {
        elg! { RoundedRectangle(3).fill(.accent).frame(height: 6) }.into_view()
    }
    fn make_thumb(&self, slider: SliderConfiguration) -> View {
        elg! {
            RoundedRectangle(2)
                .fill(.accent)
                .frame(4, 16)
                .scale_effect(move || if slider.is_pressed() { 1.25 } else { 1.0 })
        }
        .into_view()
    }
}
```

When the value changes inside `with_animation`, the parts glide to their new places.

## Row styles

A `RowStyle` draws each row of a list that people can [select in](../lists/#selecting-rows). It's given
a `RowConfiguration`: the row's `content` (the view you made for the item), the list's `axis`
(rows usually stretch across it), and whether the row `is_selected`, `is_hovered`, `is_enabled`,
and `is_focused` (the row the arrow keys are on, while the list has keyboard focus). The row still
handles clicks, keys, and screen readers itself. This one shows a bar at the leading edge instead
of a highlight:

```cpp
struct BarRowStyle : RowStyle {
    View make_body(const RowConfiguration& row) const override {
        return HStack({.spacing = 8},
                      Rectangle()
                          .fill(ThemeColor::accent)
                          .frame({.width = 3})
                          .opacity([=] { return row.is_selected() ? 1.f : 0.f; }),
                      row.content, Spacer())
            .padding(EdgeInsets::symmetric(0, 6))
            .background([=](const EnvironmentValues& env) {
                return row.is_hovered() ? env.color(ThemeColor::outline).opacity(0.4f) : Color::clear;
            });
    }
};

ScrollView(LazyVStack({.spacing = 0}, ForEach(notes, &Note::id, NoteRow))
               .selection(current.binding())
               .row_style(BarRowStyle{}));
```

```rust
struct BarRowStyle;

impl RowStyle for BarRowStyle {
    fn make_body(&self, row: RowConfiguration) -> View {
        let (selected, hovered) = (row.clone(), row.clone());
        elg! {
            HStack(spacing: 8) {
                Rectangle()
                    .fill(.accent)
                    .frame(width: 3)
                    .opacity(move || if selected.is_selected() { 1.0 } else { 0.0 })
                { row.content() }
                Spacer()
            }
            .padding(horizontal: 0, vertical: 6)
            .background(Prop::from_env(move |env| {
                ShapeStyle::Color(if hovered.is_hovered() { env.color(ThemeColor::Outline).opacity(0.4) } else { Color::CLEAR })
            }))
        }
        .into_view()
    }
}

# #[derive(Clone, PartialEq)] struct Note { id: i64, title: String }
# let notes = State::new(Vec::<Note>::new());
# let current = State::new(Option::<i64>::None);
let _ = elg! {
    ScrollView {
        LazyVStack(spacing: 0) {
            ForEach(notes, id: |n| n.id) { note in Text(note.title.clone()) }
        }
        .selection(current)
        .row_style(BarRowStyle)
    }
};
```

## Text field styles

A `TextFieldConfiguration` has the editable `field`, which you put where the text should go, and
tells you whether it `is_focused`, `is_hovered`, and `is_enabled`:

```cpp
struct UnderlinedFieldStyle : TextFieldStyle {
    View make_body(const TextFieldConfiguration& field) const override {
        return field.field.padding(EdgeInsets{.top = 6, .bottom = 6}).overlay(
            Rectangle()
                .fill([=](const EnvironmentValues& env) {
                    return field.is_focused() ? env.color(ThemeColor::accent) : env.color(ThemeColor::outline);
                })
                .frame({.height = 1}),
            Alignment::bottom);
    }
};
```

```rust
struct UnderlinedFieldStyle;

impl TextFieldStyle for UnderlinedFieldStyle {
    fn make_body(&self, field: TextFieldConfiguration) -> View {
        let focused = field.clone();
        let line = Rectangle()
            .fill(Prop::from_env(move |env| {
                ShapeStyle::Color(env.color(if focused.is_focused() { ThemeColor::Accent } else { ThemeColor::Outline }))
            }))
            .frame_with(|o| o.height(1));
        elg! {
            { field.field() }
                .padding(top: 6, bottom: 6)
                .overlay(line, alignment: .bottom)
        }
        .into_view()
    }
}
```

## Popover, tooltip and menu styles

These draw the panel around what pops up. A `PopoverConfiguration` has the `content`, and whether
it `is_padded`; a `TooltipConfiguration` has the `content`; and a `MenuConfiguration` has the
menu's `items`, already stacked. The style that applies is the one where the view that opens the
popup is, so one `.popover_style(...)` near the top styles every popover in the window. The plain
styles draw nothing around the content. A style's `motion()` says how its popups open and close
(at once, unless you say otherwise); see [Menus and popups](../menus/#opening-and-closing).

```cpp
struct CardPopoverStyle : PopoverStyle {
    View make_body(const PopoverConfiguration& popover) const override {
        return popover.content.padding(16)
            .background(ThemeColor::surface_raised, RoundedRectangle(16))
            .shadow({.color = Color::black.opacity(0.2f), .radius = 24, .y = 8});
    }
};

Root().popover_style(CardPopoverStyle{}).tooltip_style(PlainTooltipStyle{});
```

```rust
struct CardPopoverStyle;

impl PopoverStyle for CardPopoverStyle {
    fn make_body(&self, popover: PopoverConfiguration) -> View {
        elg! {
            { popover.content() }
                .padding(16)
                .background_in(.surface_raised, RoundedRectangle(16))
                .shadow(color: Color::BLACK.opacity(0.2), radius: 24, y: 8)
        }
        .into_view()
    }
}

let _ = elg! {
    VStack { Text("Root") }
        .popover_style(CardPopoverStyle)
        .tooltip_style(PlainTooltipStyle)
};
```

A `MenuStyle` also draws each row, with `make_item`. It's given a `MenuItemConfiguration`: the
row's `title`, what goes before it (`leading`: the check mark and icon) and after it (`trailing`:
the shortcut, or a submenu's arrow), each empty when the row has none, and whether the row
`is_highlighted` (under the pointer, or chosen with the keyboard), `is_enabled`, and
`is_destructive`. The row still handles clicks, keys, and submenus. By default, it's the parts in a
line on a rounded highlight; override it for taller rows or a different highlight:

```cpp
struct RoomyMenuStyle : MenuStyle {
    View make_body(const MenuConfiguration& menu) const override {
        return DefaultMenuStyle().make_body(menu);
    }
    View make_item(const MenuItemConfiguration& item) const override {
        return HStack({.spacing = 10}, item.leading, item.title, Spacer(32), item.trailing)
            .padding(EdgeInsets::symmetric(12, 9))
            .frame({.max_width = infinity, .alignment = Alignment::leading})
            .foreground_style([=](const EnvironmentValues& env) {
                return env.color(item.is_highlighted() ? ThemeColor::on_accent : ThemeColor::primary);
            })
            .background([=](const EnvironmentValues& env) {
                return item.is_highlighted() ? env.color(ThemeColor::accent) : Color::clear;
            }, RoundedRectangle(6));
    }
};
```

```rust
struct RoomyMenuStyle;

impl MenuStyle for RoomyMenuStyle {
    fn make_body(&self, menu: MenuConfiguration) -> View {
        DefaultMenuStyle.into_menu_style().make_body(&menu)
    }
    fn make_item(&self, item: MenuItemConfiguration) -> View {
        let (fg, bg) = (item.clone(), item.clone());
        elg! {
            HStack(spacing: 10) {
                { item.leading() }
                { item.title() }
                Spacer(min_length: 32)
                { item.trailing() }
            }
            .padding(horizontal: 12, vertical: 9)
            .frame(max_width: .infinity, alignment: .leading)
            .foreground_style(Prop::from_env(move |env| {
                env.color(if fg.is_highlighted() { ThemeColor::OnAccent } else { ThemeColor::Primary })
            }))
            .background_in(
                Prop::from_env(move |env| {
                    ShapeStyle::Color(if bg.is_highlighted() { env.color(ThemeColor::Accent) } else { Color::CLEAR })
                }),
                RoundedRectangle(6),
            )
        }
        .into_view()
    }
}
```
